model User {
  id                       Int                        @id @default(autoincrement())
  email                    String                     @unique
  name                     String
  last_name                String
  email_verified_at        DateTime?
  password                 String
  remember_token           String?
  phone_number             String
  phone_number_verified_at DateTime?
  num_doc                  String?
  type_doc                 DocType?
  role_default             String                     @default("CUSTOMER")
  company_id               Int?
  auth_method              String?
  provider_id              String?
  firebase_uid             String?                    @unique
  lang_user                String?
  verified_user                  Int                        @default(0)
  email_verification_code        String?
  email_verification_expires_at  DateTime?
  status                         Int                        @default(0)
  createdAt                DateTime                   @default(now())
  updatedAt                DateTime                   @updatedAt
  role                     ROLES                      @default(CUSTOMER)
  fcm_token                String?
  fcm_updated              DateTime?
}



model Countries {
  id               Int        @id @default(autoincrement())
  name_country     String
  code             String
  phone_code       String
  currency         String
  language_default String
  createdAt        DateTime   @default(now())

}



model Company {
  id                   Int                @id @default(autoincrement())
  user_id_register     Int?

  website              String?
  address              String?
  country_id           Int?
  state_id             Int?
  city_id              Int?

  company_description  String?
  company_email        String?            @unique
  company_logo         String?
  company_phone_number String?
